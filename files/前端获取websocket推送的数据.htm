2. 在.vue界面获取

定义变量（ws、重连定时器、标志位）

  ↓

connectWebSocket() → 建立 WebSocket 连接

  ├─ onopen：连接成功

  ├─ onmessage：处理收到的消息（核心逻辑）

  ├─ onclose：断线后启动重连定时器

  └─ onerror：打印错误

  ↓

onMounted → 初始化连接

  ↓

onUnmounted → 清理连接和定时器

let wsReconnectTimer = null

      console.log('[WebSocket] 收到通知消息333:', msg)

 

3.       if (Array.isArray(msg) && msg.length > 0) {

4.         // type=0 合并，type=1 独立弹窗

  ws.onclose = (e) => {

    console.log('[WebSocket] 已断开:', e.code, e.reason)

    ws = null

    // 断线 3 秒后重连

    wsReconnectTimer = setTimeout(() => {

      connectWebSocket()

    }, 3000)

  }

为什么用 setTimeout而不是 setInterval？

只用重连一次（3 秒后尝试一次）

如果重连失败，onclose会再次触发，重新设置定时器

形成 “断线 → 等 3 秒 → 重连 → 再断 → 再等 3 秒”​ 的循环

比 setInterval更安全，不会堆积重连请求

 

  ws.onerror = (e) => {

    console.error('[WebSocket] 连接错误:', e)

  }

}

 

// 初始化

onMounted(() => {

  connectWebSocket()

})

 

// 清理

onUnmounted(() => {

  if (ws) {

    ws.onclose = null // 移除重连回调

    ws.close()

    ws = null

  }

  if (wsReconnectTimer) {

    clearTimeout(wsReconnectTimer)

    wsReconnectTimer = null

  }

})
