1. 主应用vue2 引入无界

npm install wujie-vue2   # Vue2 专用包

⚠️ Vue2 项目用 wujie-vue2，Vue3 主应用才用 wujie-vue

2.  main.js 注册无界

import WujieVue from 'wujie-vue2'

Vue.use(WujieVue)

3. 新建微前端宿主组件（新增文件）

 

<!-- src/views/vue3App/index.vue -->

<template>

  <div class="wujie-wrap">

    <WujieVue

      name="ruoyi-vue3"

      :url="vue3Url"

      :props="wujieProps"

      :alive="true"

      :sync="true"

      width="100%"

      height="100%"

      @mount="onMount"

    />

  </div>

</template>

 

<script>

import { getToken } from '@/utils/auth'

 

export default {

  name: 'Vue3MicroApp',

  computed: {

    vue3Url() {

      // 开发环境走代理或直连，生产走同源子路径

      return process.env.NODE_ENV === 'development'

        ? 'http://192.168.1.132:80'   // 或 '/vue3-app'

        : '/vue3-app/'

    },

    wujieProps() {

      return {

        token: getToken(),           // 若依登录后存的 token

        userInfo: this.$store.state.user

      }

    }

  },

  methods: {

    onMount() {

      console.log('Vue3 若依子应用已挂载')

    }

  }

}

</script>

 

<style scoped>

.wujie-wrap {

  height: calc(100vh - 84px); /* 若依 Layout header+tabs 高度 */

  overflow: hidden;

}

</style>
